// --- VERİ --- // Derslerini buraya ekle (Video ID'lerini kendi videolarınla değiştir) const courses = [ { id: 1, title: '1. Derse Giriş', desc: 'Genel bakış ve sistem tanıtımı.', videoId: 'dQw4w9WgXcQ' }, { id: 2, title: '2. Temel Kavramlar', desc: 'Kursun temel direklerini öğrenin.', videoId: 'dQw4w9WgXcQ' }, { id: 3, title: '3. Uygulama ve Test', desc: 'İlk pratik uygulamanızı yapın.', videoId: 'dQw4w9WgXcQ' }, { id: 4, title: '4. İleri Seviye Giriş', desc: 'İleri düzey konuların temelleri.', videoId: 'dQw4w9WgXcQ' } ]; // --- DURUM YÖNETİMİ --- // Kaydedilmiş tamamlanan dersleri al let completedLessons = JSON.parse(localStorage.getItem('karga_academy_progress')) || []; let currentLessonId = courses[0].id; // Varsayılan ders // --- BAŞLANGIÇ --- document.addEventListener('DOMContentLoaded', () => { renderLessonList(); loadLesson(currentLessonId); // İlk dersi yükle updateOverallProgress(); }); // --- FONKSİYONLAR --- // 1. Sol Menüdeki Ders Listesini Oluştur (Checklist Tasarımı) function renderLessonList() { const listGroup = document.getElementById('lessonList'); listGroup.innerHTML = ''; // Temizle courses.forEach(lesson => { const isCompleted = completedLessons.includes(lesson.id); const isActive = lesson.id === currentLessonId; // Ders Butonu (Liste Öğesi) const button = document.createElement('button'); // 'completed' class'ını ekle, css zaten üzeri çizmeyi ve tık'ı yönetiyor button.className = `list-group-item list-group-item-action d-flex align-items-center gap-3 ${isActive ? 'active' : ''} ${isCompleted ? 'completed' : ''}`; button.setAttribute('data-id', lesson.id); // Sadece başlığa tıklanınca ders yüklensin, tık'a tıklayınca değişmesin button.innerHTML = `
${lesson.title} `; listGroup.appendChild(button); }); } // 2. Seçilen Dersi Yükle (Sağ Paneli Güncelle) function loadLesson(id) { const lesson = courses.find(c => c.id === id); if (!lesson) return; currentLessonId = id; // Video Player Güncelle document.getElementById('videoPlayer').src = `https://www.youtube.com/embed/${lesson.videoId}`; // Başlık ve Açıklama Güncelle document.getElementById('lessonTitle').innerText = lesson.title; document.getElementById('lessonDesc').innerText = lesson.desc; // Sol menüde aktifliği (vurguyu) güncelle document.querySelectorAll('#lessonList button').forEach(btn => btn.classList.remove('active')); document.querySelector(`#lessonList button[data-id="${id}"]`).classList.add('active'); } // 3. TIK (Checkbox) Durumunu Değiştir (Karga Akademi Stili) function toggleCheckStatus(id) { // Tıklandıysa kaldır, tıklanmadıysa ekle if (!completedLessons.includes(id)) { completedLessons.push(id); } else { completedLessons = completedLessons.filter(lessonId => lessonId !== id); } // LocalStorage'a kaydet (KALICILIK) localStorage.setItem('karga_academy_progress', JSON.stringify(completedLessons)); // Menüyü ve genel ilerlemeyi yenile renderLessonList(); updateOverallProgress(); // Hangi ders seçiliyse onu yeniden yükle (aktif durumunu korumak için) loadLesson(currentLessonId); } // 4. Genel İlerleme Çubuğunu Güncelle function updateOverallProgress() { const totalLessons = courses.length; const completedCount = completedLessons.length; const progressPercent = totalLessons === 0 ? 0 : Math.round((completedCount / totalLessons) * 100); document.getElementById('mainProgress').style.width = `${progressPercent}%`; document.getElementById('progressText').innerText = `${progressPercent}%`; }
